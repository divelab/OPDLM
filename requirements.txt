# Install order from a fresh conda env:
#
#   conda create -n opdlm python=3.10.19 -y
#   conda activate opdlm
#
#   # Install build tooling
#   uv pip install --upgrade setuptools wheel packaging ninja
#
#   # Install PyTorch separately first
#   python -m pip install torch==2.7.1 \
#       --index-url https://download.pytorch.org/whl/cu126
#
#   # Install OPDLM dependencies
#   uv pip install -r requirements.txt
#
#   # Install flash-attn LAST, after torch is available
#   uv pip install flash-attn==2.7.4.post1 --no-build-isolation
#
